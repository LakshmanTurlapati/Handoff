       export const TARGET_SAMPLE_RATE = 16000;
       export const FRAME_DURATION_MS = 20;
       export const FRAME_SAMPLES = TARGET_SAMPLE_RATE * FRAME_DURATION_MS / 1000; // 320
       export const SOURCE_SAMPLE_RATE = 48000;
       export const FRAME_SAMPLES_48K = SOURCE_SAMPLE_RATE * FRAME_DURATION_MS / 1000; // 960
       export const DOWNSAMPLE_WORKLET_NAME = "achilles-downsample-processor";

       export function downsample48kTo16kInt16(input: Float32Array): Int16Array {
         if (input.length !== FRAME_SAMPLES_48K) {
           throw new TypeError(`expected Float32Array length ${FRAME_SAMPLES_48K} (FRAME_SAMPLES_48K); got ${input.length}`);
         }
         // Decimation ratio is exactly 3 (48000 / 16000). For each output
         // sample we apply a simple 3-tap box-filter (sum of 3 consecutive
         // input samples divided by 3) which is a low-pass implementation
         // good enough for the Scribe v2 model. A future plan can swap in
         // a biquad anti-alias filter if pitch artefacts appear under load.
         const out = new Int16Array(FRAME_SAMPLES);
         for (let i = 0; i < FRAME_SAMPLES; i++) {
           const j = i * 3;
           const v = (input[j]! + input[j + 1]! + input[j + 2]!) / 3;
           const clamped = Math.max(-1, Math.min(1, v));
           // Float32 [-1, 1] -> Int16 [-32768, 32767]. Use 32767 for both
           // edges to keep symmetric (the asymmetric -32768 edge is the
           // CD-audio standard but pushes a 1 LSB DC offset).
           out[i] = Math.round(clamped * 32767);
         }
         return out;
       }
